#!/bin/bash
#SBATCH --account=PAS1046
#SBATCH --job-name=nlr_00_master
#SBATCH --nodes=1
#SBATCH --cpus-per-task=2
#SBATCH --mem=16G
#SBATCH --time=0:30:00
#SBATCH --mail-type=FAIL
#SBATCH --output=/fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/logs/slurm-00-%j.out

set -euo pipefail

source /fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/scripts/nlr_config.sh
for f in "${GENOME_FILE}" "${PROTEINS_FILE}" "${GFF_FILE}"; do
  if [ ! -s "${f}" ]; then
    echo "ERROR: Required input missing/empty: ${f}"
    exit 1
  fi
done

STAGE0_DIR="$(stage_dir 0)"
mkdir -p "${STAGE0_DIR}"

module load python/3.12 || true
module load samtools/1.21 || true

echo "=== Stage 0: Build master table ==="
echo "Start: $(date)"

# 0.1) Index genome (contig lengths)
if [ ! -f "${GENOME_FILE}.fai" ]; then
  echo "[0.1] samtools faidx genome"
  samtools faidx "${GENOME_FILE}"
fi
cut -f1,2 "${GENOME_FILE}.fai" > "${STAGE0_DIR}/contig_lengths.tsv"

# 0.2) Clean protein FASTA (IDs only) + lengths
echo "[0.2] Protein FASTA clean + lengths"
python3 - <<PY
import sys
from pathlib import Path

proteins = Path("${PROTEINS_FILE}")
out_faa = Path("${STAGE0_DIR}/proteins_clean.faa")
out_len = Path("${STAGE0_DIR}/protein_lengths.tsv")

def fasta_iter(fp):
    name=None; seq=[]
    for line in fp:
        line=line.rstrip("\\n")
        if not line: 
            continue
        if line.startswith(">"):
            if name:
                yield name, "".join(seq)
            name=line[1:].split()[0]
            seq=[]
        else:
            seq.append(line)
    if name:
        yield name, "".join(seq)

with proteins.open() as f, out_faa.open("w") as o, out_len.open("w") as l:
    l.write("protein_id\\tprotein_length\\n")
    for pid, seq in fasta_iter(f):
        o.write(f">{pid}\\n")
        for i in range(0, len(seq), 60):
            o.write(seq[i:i+60] + "\\n")
        l.write(f"{pid}\\t{len(seq)}\\n")
PY

# 0.3) Parse GFF3: transcript/protein IDs and coords
# For funannotate, protein IDs usually match transcript IDs in proteins.fa headers.
echo "[0.3] Parse GFF3 mRNA features -> coords"
python3 - <<PY
import re
import pandas as pd

gff = "${GFF_FILE}"
rows = []

def parse_attrs(attr_str):
    d={}
    for item in attr_str.split(";"):
        if "=" in item:
            k,v=item.split("=",1)
            d[k]=v
    return d

with open(gff) as f:
    for line in f:
        if not line.strip() or line.startswith("#"):
            continue
        parts=line.rstrip("\\n").split("\\t")
        if len(parts) < 9:
            continue
        seqid, source, ftype, start, end, score, strand, phase, attrs = parts
        if ftype not in ("mRNA","transcript"):
            continue
        A = parse_attrs(attrs)
        tid = A.get("ID","")
        gid = A.get("Parent","")
        if not tid:
            continue
        rows.append({
            "protein_id": tid,
            "transcript_id": tid,
            "gene_id": gid,
            "scaffold": seqid,
            "start": int(start),
            "end": int(end),
            "strand": strand
        })

df = pd.DataFrame(rows).drop_duplicates()
df.to_csv("${STAGE0_DIR}/id_coords.tsv", sep="\\t", index=False)
print(f"Parsed {len(df)} transcript entries")
PY

# 0.4) Parse eggNOG (light columns) and merge into master
echo "[0.4] Parse eggNOG + merge (fail-soft if missing)"
python3 - <<PY
import os
import pandas as pd

lens = pd.read_csv("${STAGE0_DIR}/protein_lengths.tsv", sep="\\t")
coords = pd.read_csv("${STAGE0_DIR}/id_coords.tsv", sep="\\t")

eggnog_path = "${EGGNOG_FILE}"
eg_rows = []

# Always write a lightweight eggnog table for provenance/debugging
eggnog_light_out = "${STAGE0_DIR}/eggnog_light.tsv"

if os.path.exists(eggnog_path) and os.path.getsize(eggnog_path) > 0:
    with open(eggnog_path) as f:
        for line in f:
            if line.startswith("#") or not line.strip():
                continue
            cols = line.rstrip("\\n").split("\\t")
            q = cols[0]
            rec = {"protein_id": q}
            if len(cols) > 6:  rec["COG_category"]    = cols[6]
            if len(cols) > 7:  rec["Description"]     = cols[7]
            if len(cols) > 8:  rec["Preferred_name"]  = cols[8]
            if len(cols) > 9:  rec["GOs"]             = cols[9]
            if len(cols) > 11: rec["KEGG_ko"]         = cols[11]
            eg_rows.append(rec)
else:
    # eggNOG not available yet: keep empty but with expected columns
    eg_rows = []

eg = pd.DataFrame(eg_rows)
if eg.empty:
    eg = pd.DataFrame(columns=["protein_id","COG_category","Description","Preferred_name","GOs","KEGG_ko"])

eg.drop_duplicates().to_csv(eggnog_light_out, sep="\\t", index=False)

master = coords.merge(lens, on="protein_id", how="left").merge(eg, on="protein_id", how="left")
master.to_csv("${STAGE0_DIR}/master_table.tsv", sep="\\t", index=False)

print("Master table rows:", master.shape[0])
print("eggNOG light rows:", eg.shape[0])
print("Wrote:", eggnog_light_out)
PY

echo "Done: $(date)"

#!/bin/bash
#SBATCH --account=PAS1046
#SBATCH --job-name=nlr_01_discover

#SBATCH --output=/fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/logs/slurm-01-%j.out

set -euo pipefail
source /fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/scripts/nlr_config.sh

# Basic input checks

  echo "ERROR: Proteins FASTA missing/empty: ${PROT}"

  echo "WARN: master_table.tsv missing/empty: ${MASTER}"

echo "[1.2] Parse NBD hits (hard-filter i-Evalue + aligned length)"
python3 - <<PY
import pandas as pd

DOMTBL = "${STAGE1_DIR}/nbd_whole.domtblout"
EVAL = float("${EVAL_NBD}")
MIN_ALI = int("${MIN_NBD_ALI_LEN}")

rows=[]
total=0
short=0
bade=0
parsed=0

with open(DOMTBL) as f:
    for line in f:
        if line.startswith("#"):
            continue
        total += 1
        x=line.split()
        # domtblout has fixed first 22 columns; description may extend beyond
        if len(x) < 22:
            continue

        hmm  = x[0]      # target name
        prot = x[3]      # query name
        try:
            i_eval = float(x[12])  # domain i-Evalue
            ali_start = int(x[17])
            ali_end   = int(x[18])
        except Exception:
            continue

        alilen = abs(ali_end - ali_start) + 1
        if i_eval > EVAL:
            bade += 1
            continue
        if alilen < MIN_ALI:
            short += 1
            continue

        parsed += 1
        rows.append((prot, hmm, i_eval, ali_start, ali_end, alilen))

df=pd.DataFrame(rows, columns=["protein_id","nbd_hmm","dom_i_evalue","ali_start","ali_end","ali_len"])
df.to_csv("${STAGE1_DIR}/nbd_hits.tsv", sep="\\t", index=False)

cand = df[["protein_id"]].drop_duplicates()
cand.to_csv("${STAGE1_DIR}/nbd_candidate_ids.txt", index=False, header=False)

with open("${STAGE1_DIR}/nbd_parse_report.txt","w") as out:
    out.write(f"total_domtbl_lines\\t{total}\\n")
    out.write(f"kept_after_filters\\t{parsed}\\n")
    out.write(f"filtered_high_i_eval\\t{bade}\\n")
    out.write(f"filtered_short_align\\t{short}\\n")
    out.write(f"unique_candidate_proteins\\t{cand.shape[0]}\\n")

print("NBD candidate proteins:", cand.shape[0])
print("Wrote:", "${STAGE1_DIR}/nbd_parse_report.txt")
PY

# 1.3) eggNOG priority list (NOT gating)
echo "[1.3] eggNOG priority list (optional, not gating)"
python3 - <<PY
import pandas as pd, re, os

MASTER="${MASTER}"
OUT="${STAGE1_DIR}/eggnog_priority_ids.txt"

if (not os.path.exists(MASTER)) or os.path.getsize(MASTER)==0:
    open(OUT,"w").close()
    print("MASTER missing/empty; wrote empty:", OUT)
    raise SystemExit(0)

master = pd.read_csv(MASTER, sep="\\t")
if "protein_id" not in master.columns:
    open(OUT,"w").close()
    print("MASTER lacks protein_id column; wrote empty:", OUT)
    raise SystemExit(0)

cols = [c for c in ["Description","Preferred_name","GOs"] if c in master.columns]

kw = [
  "NACHT","NB-ARC","STAND","NOD-like",
  "WD40","ANK","TPR","HEAT","Kelch",
  "HET","gasdermin","HeLo","MLKL","Goodbye",
  "inflammasome","immune"
]
pat = re.compile("|".join([re.escape(k) for k in kw]), re.IGNORECASE)

def row_hit(r):
    for c in cols:
        v = r.get(c,"")
        if isinstance(v,str) and pat.search(v):
            return True
    return False

if cols:
    prio = master[master.apply(row_hit, axis=1)][["protein_id"]].dropna().drop_duplicates()
else:
    prio = master[["protein_id"]].head(0)

prio.to_csv(OUT, index=False, header=False)
print("eggNOG priority proteins:", prio.shape[0])
PY

# 1.4) UNION for architecture scanning = NBD hits ∪ eggNOG priority

#!/bin/bash
#SBATCH --account=PAS1046
#SBATCH --job-name=nlr_02_arch
#SBATCH --nodes=1
#SBATCH --cpus-per-task=16
#SBATCH --mem=64G
#SBATCH --time=00:30:00
#SBATCH --mail-type=FAIL
#SBATCH --output=/fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/logs/slurm-02-%j.out

set -euo pipefail
source /fs/ess/PAS1046/projects/novogene_data_2024ff/110C_mushrooms/2024_05_28/Pc_BM/EGAP2/nlr_discovery/scripts/nlr_config.sh

STAGE1_DIR="$(stage_dir 1)"
STAGE2_DIR="$(stage_dir 2)"
mkdir -p "${STAGE2_DIR}"
cd "${STAGE2_DIR}"

module load python/3.12 || true

Q="${STAGE1_DIR}/union_candidates.faa"
NBD_IDS="${STAGE1_DIR}/nbd_candidate_ids.txt"

if [ ! -s "${Q}" ]; then
  echo "ERROR: union candidates FASTA missing/empty: ${Q}"
  exit 1
fi

if [ ! -s "${NBD_IDS}" ]; then
  echo "WARN: Stage1 NBD id list missing/empty: ${NBD_IDS}"
  echo "      Stage2 will still run, but NBD presence may be underestimated for some proteins."
  : > "${STAGE2_DIR}/stage1_nbd.ids"
else
  sort -u "${NBD_IDS}" > "${STAGE2_DIR}/stage1_nbd.ids"
fi

echo "=== Stage 2: Domain architecture ==="
echo "Start: $(date)"
echo "STAGE1_DIR: ${STAGE1_DIR}"
echo "STAGE2_DIR: ${STAGE2_DIR}"
echo "Q: ${Q}"
echo "NBD_IDS: ${NBD_IDS}"
echo "PFAM_DB: ${PFAM_DB}"
echo "CUSTOM_HMMS: ${CUSTOM_HMMS}"
echo "EVAL_PFAM: ${EVAL_PFAM}"
echo "SLURM_CPUS_PER_TASK: ${SLURM_CPUS_PER_TASK:-unset}"

EVAL_USE="${EVAL_PFAM}"
if [ "${MAX_DOM_I_EVAL}" != "0" ]; then
  EVAL_USE="${MAX_DOM_I_EVAL}"
fi

echo "[2.1] Pfam scan"

if [ ! -f "${PFAM_DB}" ]; then
  echo "ERROR: PFAM_DB not found: ${PFAM_DB}"
  exit 1
fi

for ext in h3m h3i h3f h3p; do
  if [ ! -f "${PFAM_DB}.${ext}" ]; then
    echo "ERROR: Missing pressed PFAM file: ${PFAM_DB}.${ext}"
    echo "Fix once: apptainer exec --bind /fs/ess/PAS1046 ${HMMER_CONTAINER} hmmpress -f ${PFAM_DB}"
    exit 1
  fi
done

apptainer exec --bind /fs/ess/PAS1046 "${HMMER_CONTAINER}" \
  hmmscan --cpu "${SLURM_CPUS_PER_TASK}" \
  --domtblout "${STAGE2_DIR}/pfam.domtblout" \
  --incE "${EVAL_PFAM}" --incdomE "${EVAL_PFAM}" \
  "${PFAM_DB}" "${Q}" > "${STAGE2_DIR}/pfam.hmmscan.txt"

echo "[2.2] Custom scan (optional)"

if [ -f "${CUSTOM_HMMS}" ]; then
  for ext in h3m h3i h3f h3p; do
    if [ ! -f "${CUSTOM_HMMS}.${ext}" ]; then
      echo "ERROR: Missing pressed custom HMM file: ${CUSTOM_HMMS}.${ext}"
      echo "Fix once: apptainer exec --bind /fs/ess/PAS1046 ${HMMER_CONTAINER} hmmpress -f ${CUSTOM_HMMS}"
      exit 1
    fi
  done

  apptainer exec --bind /fs/ess/PAS1046 "${HMMER_CONTAINER}" \
    hmmscan --cpu "${SLURM_CPUS_PER_TASK}" \
    --domtblout "${STAGE2_DIR}/custom.domtblout" \
    --incE "${EVAL_PFAM}" --incdomE "${EVAL_PFAM}" \
    "${CUSTOM_HMMS}" "${Q}" > "${STAGE2_DIR}/custom.hmmscan.txt"
else
  echo "Custom HMM file not found (skipping): ${CUSTOM_HMMS}"
  : > "${STAGE2_DIR}/custom.domtblout"
fi

echo "[2.3] Parse into per-protein architectures (keep ALL proteins) + soft FP flags"
python3 - <<PY
import os, re
import pandas as pd

FASTA = "${Q}"
EVAL = float("${EVAL_USE}")
MIN_ALI = int("${MIN_DOM_ALI_LEN}")
STAGE1_NBD = "${STAGE2_DIR}/stage1_nbd.ids"

def fasta_ids(path):
    ids=[]
    with open(path) as f:
        for line in f:
            if line.startswith(">"):
                ids.append(line[1:].split()[0])
    return ids

all_prots = fasta_ids(FASTA)
all_prot_set = set(all_prots)

stage1_nbd = set()
if os.path.exists(STAGE1_NBD) and os.path.getsize(STAGE1_NBD) > 0:
    with open(STAGE1_NBD) as f:
        for line in f:
            stage1_nbd.add(line.strip())

def parse_domtbl(path, source):
    rows=[]
    total=0; kept=0; bade=0; short=0
    if not os.path.exists(path) or os.path.getsize(path) == 0:
        return pd.DataFrame(columns=["protein_id","domain","acc","dom_i_evalue","start","end","ali_len","source"]), {
            "source": source, "total_domtbl_lines": 0, "kept_after_filters": 0,
            "filtered_high_i_eval": 0, "filtered_short_align": 0
        }
    with open(path) as f:
        for line in f:
            if line.startswith("#"): continue
            total += 1
            x=line.split()
            if len(x) < 22: continue

import re
import pandas as pd
import sys
import os

# Read shell variables directly
gff = "${MINIPROT_GFF}"
features_out = "${TABLES_DIR}/miniprot_features.tsv"
summary_out = "${TABLES_DIR}/miniprot_summary.tsv"
counts_out = "${TABLES_DIR}/miniprot_hit_counts.tsv"

print(f"Reading GFF: {gff}", file=sys.stderr)

attr_re = re.compile(r"([A-Za-z0-9_]+)=([^;]+)")

def attrs(s):
    d={}
    for m in attr_re.finditer(s):
        d[m.group(1)] = m.group(2)
    return d

rows=[]
with open(gff) as f:
    for line in f:
        if line.startswith("#") or not line.strip():
            continue
        parts = line.rstrip("\n").split("\t")
        if len(parts) != 9:
            continue
        chrom,src,ft,start,end,score,strand,phase,attr = parts
        A = attrs(attr)

        # miniprot commonly uses Target=QUERY ...
        q = ""
        if "Target" in A:
            q = A["Target"].split()[0]
        elif "query" in A:
            q = A["query"]
        elif "Q" in A:
            q = A["Q"]
        elif "ID" in A:
            q = A["ID"]

        rows.append({
            "protein_id": q,
            "scaffold": chrom,
            "start": int(start),
            "end": int(end),
            "strand": strand,
            "feature": ft,
            "source": src,
            "score": score,
            "phase": phase,
            "attrs_raw": attr
        })

df = pd.DataFrame(rows)
df.to_csv(features_out, sep="\t", index=False)

if df.empty:
    pd.DataFrame(columns=["protein_id","scaffold","start","end","strand"]).to_csv(summary_out, sep="\t", index=False)
    pd.DataFrame(columns=["feature","count"]).to_csv(counts_out, sep="\t", index=False)
    print("Parsed 0 features.", file=sys.stderr)
else:
    # one locus per protein: min(start), max(end) across all features
    loc = df.groupby("protein_id", dropna=False).agg(
        scaffold=("scaffold","first"),
        start=("start","min"),
        end=("end","max"),
        strand=("strand","first"),
        n_features=("feature","size")
    ).reset_index()

    loc.to_csv(summary_out, sep="\t", index=False)

    counts = df["feature"].value_counts().rename_axis("feature").reset_index(name="count")
    counts.to_csv(counts_out, sep="\t", index=False)

    print(f"Parsed features: {df.shape[0]}", file=sys.stderr)
    print(f"Proteins with loci: {loc.shape[0]}", file=sys.stderr)
PY

import os
import re
import pandas as pd
import sys

# Read shell variables directly
merged = "${MERGED}"
summary_out = "${TABLES_DIR}/exonerate_summary.tsv"
counts_out = "${SUMM_DIR}/exonerate_counts.tsv"

# but it is tab-delimited with 9 fields for feature lines.
with open(merged) as f:
    for line in f:
        if line.startswith("#") or not line.strip():
            continue
        parts = line.rstrip("\n").split("\t")
        if len(parts) != 9:
            continue
        chrom, source, feature, start, end, score, strand, phase, attrs = parts

        # Try to recover query/protein id from attributes.
        # Exonerate commonly includes something like: "gene_id <id>" or similar.
        # We do best-effort parsing:
        pid = ""
        m = re.search(r"(query|Query|protein|prot|sequence|seq)[ _:=]+([A-Za-z0-9_.-]+)", attrs)
        if m:
            pid = m.group(2)
        else:
            # fallback: look for an identifier-like token
            m2 = re.search(r"([A-Za-z0-9_.-]+)", attrs)
            pid = m2.group(1) if m2 else ""

        try:
            s = int(start); e = int(end)
        except Exception:
            continue

        rows.append({
            "protein_id": pid,
            "scaffold": chrom,
            "start": s,
            "end": e,
            "strand": strand,
            "feature": feature,
            "score": score,
            "source": source
        })

df = pd.DataFrame(rows)
if df.empty:
    pd.DataFrame(columns=["protein_id","scaffold","start","end","strand","n_features"]).to_csv(summary_out, sep="\t", index=False)
    pd.DataFrame(columns=["feature","count"]).to_csv(counts_out, sep="\t", index=False)
    print("Parsed 0 features.", file=sys.stderr)
else:
    loc = df.groupby("protein_id", dropna=False).agg(
        scaffold=("scaffold","first"),
        start=("start","min"),
        end=("end","max"),
        strand=("strand","first"),
        n_features=("feature","size")
    ).reset_index()

    loc.to_csv(summary_out, sep="\t", index=False)
    df["feature"].value_counts().rename_axis("feature").reset_index(name="count").to_csv(counts_out, sep="\t", index=False)
    print(f"Parsed rows: {df.shape[0]}", file=sys.stderr)
    print(f"Proteins summarized: {df['protein_id'].nunique()}", file=sys.stderr)

def normalize_bool_series(s):
    # Accepts bool/int/str-ish; returns boolean series with NA -> False
    if s is None:
        return None
    if s.dtype == bool:
        return s.fillna(False)
    if np.issubdtype(s.dtype, np.number):
        return (s.fillna(0) != 0)
    # strings like TRUE/FALSE/yes/no/1/0
    x = s.astype(str).str.strip().str.lower()
    return x.isin(["true","t","yes","y","1","ok","pass"])

def find_col(df, candidates):
    for c in candidates:
        if c in df.columns:
            return c
    # case-insensitive match
    low = {c.lower(): c for c in df.columns}
    for cand in candidates:
        if cand.lower() in low:
            return low[cand.lower()]
    return None
